       327 +        T::Slash | T::Asterisk => Prec::Product,             
       328 +        T::LParen => Prec::Call,                             
       329 +        T::LBracket => Prec::Index,                          
       330 +        _ => Prec::Lowest,                                   
       331 +    }                                                        
       332 +}                                                            
       333 +                                                             
       334 +struct Parser {                                              
       335 +    toks: Vec<Token>,                                        
       336 +    pos: usize,                                              
       337 +    errors: Vec<String>,                                     
       338 +}                                                            
       339 +                                                             
       340 +impl Parser {                                                
       341 +    fn cur(&self) -> &Token {                                
       342 +        &self.toks[self.pos.min(self.toks.len() - 1)]        
       343 +    }                                                        
       344 +    fn peek(&self) -> &Token {                               
       345 +        &self.toks[(self.pos + 1).min(self.toks.len() - 1)]  
       346 +    }                                                        
       347 +    fn next(&mut self) {                                     
       348 +        if self.pos < self.toks.len() - 1 {                  
       349 +            self.pos += 1;                                   
       350 +        }                                                    
       351 +    }                                                        
       352 +    fn cur_is(&self, t: T) -> bool {                         
       353 +        self.cur().t == t                                    
       354 +    }                                                        
       355 +    fn peek_is(&self, t: T) -> bool {                        
       356 +        self.peek().t == t                                   
       357 +    }                                                        
       358 +    fn expect_peek(&mut self, t: T) -> bool {                
       359 +        if self.peek_is(t) {                                 
       360 +            self.next();                                     
       361 +            true                                             
       362 +        } else {                                             
       363 +            self.errors.push(format!("expected next token to 
           +be {}, got {} instead", t.name(), self.peek().t.name()));    
       364 +            false                                            
       365 +        }                                                    
       366 +    }                                                        
       367 +                                                             
       368 +    fn program(&mut self) -> Vec<Stmt> {                     
       369 +        let mut stmts = Vec::new();                          
       370 +        while !self.cur_is(T::Eof) {                         
       371 +            if let Some(s) = self.statement() {              
       372 +                stmts.push(s);                               
       373 +            }                                                
       374 +            self.next();                                     
       375 +        }                                                    
       376 +        stmts                                                
       377 +    }                                                        
       378 +                                                             
       379 +    fn statement(&mut self) -> Option<Stmt> {                
       380 +        match self.cur().t {                                 
       381 +            T::Let => {                                      
       382 +                let line = self.cur().line;                  
       383 +                if !self.expect_peek(T::Ident) {             
       384 +                    return None;                             
       385 +                }                                            
       386 +                let name = self.cur().lit.clone();           
       387 +                if !self.expect_peek(T::Assign) {            
       388 +                    return None;                             
       389 +                }                                            
       390 +                self.next();                                 
       391 +                let v = self.expression(Prec::Lowest);       
       392 +                if self.peek_is(T::Semicolon) {              
       393 +                    self.next();                             
       394 +                }                                            
       395 +                Some(Stmt::Let(name, v, line))               
       396 +            }                                                
       397 +            T::Return => {                                   
       398 +                let line = self.cur().line;                  
       399 +                self.next();                                 
       400 +                let v = self.expression(Prec::Lowest);       
       401 +                if self.peek_is(T::Semicolon) {              
       402 +                    self.next();                             
       403 +                }                                            
       404 +                Some(Stmt::Return(v, line))                  
       405 +            }                                                
       406 +            _ => {                                           
       407 +                let e = self.expression(Prec::Lowest);       
       408 +                if self.peek_is(T::Semicolon) {              
       409 +                    self.next();                             
       410 +                }                                            
       411 +                Some(Stmt::Expr(e))                          
       412 +            }                                                
       413 +        }                                                    
       414 +    }                                                        
       415 +                                                             
       416 +    fn expression(&mut self, prec: Prec) -> Expr {           
       417 +        let Some(mut left) = self.prefix() else {            
       418 +            return Expr::Missing;                            
       419 +        };                                                   
       420 +        while !self.peek_is(T::Semicolon) && prec < precedenc
           +e(self.peek().t) {                                           
       421 +            let t = self.peek().t;                           
       422 +            let line = self.peek().line;                     
       423 +            match t {                                        
       424 +                T::Plus | T::Minus | T::Slash | T::Asterisk |
           + T::Eq | T::NotEq | T::Lt | T::Gt => {                       
       425 +                    self.next();                             
       426 +                    let op = self.cur().lit.clone();         
       427 +                    let p = precedence(t);                   
       428 +                    self.next();                             
       429 +                    let right = self.expression(p);          
       430 +                    left = Expr::Infix(Box::new(left), op, Bo
           +x::new(right), line);                                        
       431 +                }                                            
       432 +                T::LParen => {                               
       433 +                    self.next();                             
       434 +                    let args = self.expression_list(T::RParen
           +);                                                           
       435 +                    left = Expr::Call(Box::new(left), args.un
           +wrap_or_default(), line);                                    
       436 +                }                                            
       437 +                T::LBracket => {                             
       438 +                    self.next();                             
       439 +                    self.next();                             
       440 +                    let idx = self.expression(Prec::Lowest); 
       441 +                    if !self.expect_peek(T::RBracket) {      
       442 +                        return Expr::Missing;                
       443 +                    }                                        
       444 +                    left = Expr::Index(Box::new(left), Box::n
           +ew(idx), line);                                              
       445 +                }                                            
       446 +                _ => return left,                            
       447 +            }                                                
       448 +        }                                                    
       449 +        left                                                 
       450 +    }                                                        
       451 +                                                             
       452 +    /// Returns `None` where the Go parser's prefix function 
           +is missing (the                                              
       453 +    /// caller then yields a nil expression); `Some(Expr::Mis
           +sing)` where a                                               
       454 +    /// prefix function itself returned nil.                 
       455 +    fn prefix(&mut self) -> Option<Expr> {                   
       456 +        let tok = self.cur().clone();                        
       457 +        Some(match tok.t {                                   
       458 +            T::Ident => Expr::Ident(tok.lit),                
       459 +            T::Int => match tok.lit.parse::<i64>() {         
       460 +                Ok(i) => Expr::Int(i),                       
       461 +                Err(_) => {                                  
       462 +                    self.errors.push(format!("could not parse
           + {:?} as integer", tok.lit));                                
       463 +                    Expr::Missing                            
       464 +                }                                            
       465 +            },                                               
       466 +            T::Str => Expr::Str(tok.lit),                    
       467 +            T::True => Expr::Bool(true),                     
       468 +            T::False => Expr::Bool(false),                   
       469 +            T::Bang | T::Minus => {                          
       470 +                self.next();                                 
       471 +                let r = self.expression(Prec::Prefix);       
       472 +                Expr::Prefix(tok.lit, Box::new(r), tok.line) 
       473 +            }                                                
       474 +            T::LParen => {                                   
       475 +                self.next();                                 
       476 +                let e = self.expression(Prec::Lowest);       
       477 +                if !self.expect_peek(T::RParen) {            
       478 +                    return Some(Expr::Missing);              
       479 +                }                                            
       480 +                e                                            
       481 +            }                                                
       482 +            T::If => {                                       
       483 +                if !self.expect_peek(T::LParen) {            
       484 +                    return Some(Expr::Missing);              
       485 +                }                                            
       486 +                self.next();                                 
       487 +                let cond = self.expression(Prec::Lowest);    
       488 +                if !self.expect_peek(T::RParen) || !self.expe
           +ct_peek(T::LBrace) {                                         
       489 +                    return Some(Expr::Missing);              
       490 +                }                                            
       491 +                let then = self.block();                     
       492 +                let mut els = None;                          
       493 +                if self.peek_is(T::Else) {                   
       494 +                    self.next();                             
       495 +                    if !self.expect_peek(T::LBrace) {        
       496 +                        return Some(Expr::Missing);          
       497 +                    }                                        
       498 +                    els = Some(self.block());                
       499 +                }                                            
       500 +                Expr::If(Box::new(cond), then, els)          
       501 +            }                                                
       502 +            T::Function | T::Macro => {                      
       503 +                if !self.expect_peek(T::LParen) {            
       504 +                    return Some(Expr::Missing);              
       505 +                }                                            
       506 +                let Some(params) = self.parameters() else {  
       507 +                    return Some(Expr::Missing);              
       508 +                };                                           
       509 +                if !self.expect_peek(T::LBrace) {            
       510 +                    return Some(Expr::Missing);              
       511 +                }                                            
       512 +                let body = self.block();                     
       513 +                if tok.t == T::Function {                    
       514 +                    Expr::Fn(params, body)                   
       515 +                } else {                                     
       516 +                    Expr::Macro(params, body)                
       517 +                }                                            
       518 +            }                                                
       519 +            T::LBracket => match self.expression_list(T::RBra
           +cket) {                                                      
       520 +                Some(xs) => Expr::Array(xs),                 
       521 +                None => Expr::Missing,                       
       522 +            },                                               
       523 +            T::LBrace => {                                   
       524 +                let mut pairs = Vec::new();                  
       525 +                while !self.peek_is(T::RBrace) {             
       526 +                    self.next();                             
       527 +                    let k = self.expression(Prec::Lowest);   
       528 +                    if !self.expect_peek(T::Colon) {         
       529 +                        return Some(Expr::Missing);          
       530 +                    }                                        
       531 +                    self.next();                             
       532 +                    let v = self.expression(Prec::Lowest);   
       533 +                    pairs.push((k, v));                      
       534 +                    if !self.peek_is(T::RBrace) && !self.expe
           +ct_peek(T::Comma) {                                          
       535 +                        return Some(Expr::Missing);          
       536 +                    }                                        
       537 +                }                                            
       538 +                if !self.expect_peek(T::RBrace) {            
       539 +                    return Some(Expr::Missing);              
       540 +                }                                            
       541 +                Expr::Hash(pairs)                            
       542 +            }                                                
       543 +            t => {                                           
       544 +                self.errors.push(format!("no prefix parse fun
           +ction for {} found", t.name()));                             
       545 +                return None;                                 
       546 +            }                                                
       547 +        })                                                   
       548 +    }                                                        
       549 +                                                             
       550 +    fn block(&mut self) -> Vec<Stmt> {                       
       551 +        let mut stmts = Vec::new();                          
       552 +        self.next();                                         
       553 +        while !self.cur_is(T::RBrace) && !self.cur_is(T::Eof)
           + {                                                           
       554 +            if let Some(s) = self.statement() {              
       555 +                stmts.push(s);                               
       556 +            }                                                
       557 +            self.next();                                     
       558 +        }                                                    
       559 +        stmts                                                
       560 +    }                                                        
       561 +                                                             
       562 +    fn parameters(&mut self) -> Option<Vec<String>> {        
       563 +        let mut ps = Vec::new();                             
       564 +        if self.peek_is(T::RParen) {                         
       565 +            self.next();                                     
       566 +            return Some(ps);                                 
       567 +        }                                                    
       568 +        self.next();                                         
       569 +        ps.push(self.cur().lit.clone());                     
       570 +        while self.peek_is(T::Comma) {                       
       571 +            self.next();                                     
       572 +            self.next();                                     
       573 +            ps.push(self.cur().lit.clone());                 
       574 +        }                                                    
       575 +        if !self.expect_peek(T::RParen) {                    
       576 +            return None;                                     
       577 +        }                                                    
       578 +        Some(ps)                                             
       579 +    }                                                        
       580 +                                                             
       581 +    fn expression_list(&mut self, end: T) -> Option<Vec<Expr>
           +> {                                                          
       582 +        let mut xs = Vec::new();                             
       583 +        if self.peek_is(end) {                               
       584 +            self.next();                                     
       585 +            return Some(xs);                                 
       586 +        }                                                    
       587 +        self.next();                                         
       588 +        xs.push(self.expression(Prec::Lowest));              
       589 +        while self.peek_is(T::Comma) {                       
       590 +            self.next();                                     
       591 +            self.next();                                     
       592 +            xs.push(self.expression(Prec::Lowest));          
       593 +        }                                                    
       594 +        if !self.expect_peek(end) {                          
       595 +            return None;                                     
       596 +        }                                                    
       597 +        Some(xs)                                             
       598 +    }                                                        
       599 +}                                                            
       600 +                                                             
       601 +pub fn parse(src: &str) -> Result<Vec<Stmt>, Vec<String>> {  
       602 +    let mut p = Parser { toks: lex(src), pos: 0, errors: Vec:
           +:new() };                                                    
       603 +    let prog = p.program();                                  
       604 +    if p.errors.is_empty() {                                 
       605 +        Ok(prog)                                             
       606 +    } else {                                                 
       607 +        Err(p.errors)                                        
       608 +    }                                                        
       609 +}                                                            
       610 +                                                             
       611 +// ----------------------------------------------------------
           +-----------------                                            
       612 +// Scope analysis                                            
       613 +                                                             
       614 +/// Names bound in a function's own environment: its paramete
           +rs and every                                                 
       615 +/// `let` in its body outside nested function literals.      
       616 +fn collect_lets(stmts: &[Stmt], out: &mut HashSet<String>) { 
       617 +    fn expr(e: &Expr, out: &mut HashSet<String>) {           
       618 +        match e {                                            
       619 +            Expr::If(c, t, el) => {                          
       620 +                expr(c, out);                                
       621 +                collect_lets(t, out);                        
       622 +                if let Some(el) = el {                       
       623 +                    collect_lets(el, out);                   
       624 +                }                                            
       625 +            }                                                
       626 +            Expr::Prefix(_, r, _) => expr(r, out),           
       627 +            Expr::Infix(l, _, r, _) | Expr::Index(l, r, _) =>
           + {                                                           
       628 +                expr(l, out);                                
       629 +                expr(r, out);                                
       630 +            }                                                
       631 +            Expr::Call(c, args, _) => {                      
       632 +                expr(c, out);                                
       633 +                args.iter().for_each(|a| expr(a, out));      
       634 +            }                                                
       635 +            Expr::Array(xs) => xs.iter().for_each(|a| expr(a,
           + out)),                                                      
       636 +            Expr::Hash(ps) => ps.iter().for_each(|(k, v)| {  
       637 +                expr(k, out);                                
       638 +                expr(v, out)                                 
       639 +            }),                                              
       640 +            _ => {}                                          
       641 +        }                                                    
       642 +    }                                                        
       643 +    for s in stmts {                                         
       644 +        match s {                                            
       645 +            Stmt::Let(n, v, _) => {                          
       646 +                out.insert(n.clone());                       
       647 +                expr(v, out);                                
       648 +            }                                                
       649 +            Stmt::Return(v, _) | Stmt::Expr(v) => expr(v, out
           +),                                                           
       650 +        }                                                    
       651 +    }                                                        
       652 +}                                                            
       653 +                                                             
       654 +/// Names that a function literal may look up in its enclosin
           +g                                                            
       655 +/// environments: identifiers used in it or in functions nest
           +ed inside it,                                                
       656 +/// except its own parameters (which are always bound).      
       657 +fn escaping(params: &[String], body: &[Stmt]) -> HashSet<Stri
           +ng> {                                                        
       658 +    fn expr(e: &Expr, out: &mut HashSet<String>) {           
       659 +        match e {                                            
       660 +            Expr::Ident(n) => {                              
       661 +                out.insert(n.clone());                       
       662 +            }                                                
       663 +            Expr::Fn(ps, b) | Expr::Macro(ps, b) => out.exten
           +d(escaping(ps, b)),                                          
       664 +            Expr::If(c, t, el) => {                          
       665 +                expr(c, out);                                
       666 +                stmts(t, out);                               
       667 +                if let Some(el) = el {                       
       668 +                    stmts(el, out);                          
       669 +                }                                            
       670 +            }                                                
       671 +            Expr::Prefix(_, r, _) => expr(r, out),           
       672 +            Expr::Infix(l, _, r, _) | Expr::Index(l, r, _) =>
           + {                                                           
       673 +                expr(l, out);                                
       674 +                expr(r, out);                                
       675 +            }                                                
       676 +            Expr::Call(c, args, _) => {                      
       677 +                expr(c, out);                                
       678 +                args.iter().for_each(|a| expr(a, out));      
       679 +            }                                                
       680 +            Expr::Array(xs) => xs.iter().for_each(|a| expr(a,
           + out)),                                                      
       681 +            Expr::Hash(ps) => ps.iter().for_each(|(k, v)| {  
       682 +                expr(k, out);                                
       683 +                expr(v, out)                                 
       684 +            }),                                              
       685 +            Expr::Missing | Expr::Int(_) | Expr::Bool(_) | Ex
           +pr::Str(_) => {}                                             
       686 +        }                                                    
       687 +    }                                                        
       688 +    fn stmts(ss: &[Stmt], out: &mut HashSet<String>) {       
       689 +        for s in ss {                                        
       690 +            match s {                                        
       691 +                Stmt::Let(_, v, _) | Stmt::Return(v, _) | Stm
           +t::Expr(v) => expr(v, out),                                  
       692 +            }                                                
       693 +        }                                                    
       694 +    }                                                        
       695 +    let mut out = HashSet::new();                            
       696 +    stmts(body, &mut out);                                   
       697 +    for p in params {                                        
       698 +        out.remove(p);                                       
       699 +    }                                                        
       700 +    out                                                      
       701 +}                                                            
       702 +                                                             
       703 +/// Names used by function literals nested (at any depth) in 
           +`body` that                                                  
       704 +/// may resolve to the enclosing function's environment.     
       705 +fn used_by_inner(body: &[Stmt]) -> HashSet<String> {         
       706 +    // `escaping` of a parameterless pseudo-function over the
           + body includes                                               
       707 +    // direct uses too; subtract nothing but collect only fro
           +m nested fns.                                                
       708 +    fn expr(e: &Expr, out: &mut HashSet<String>) {           
       709 +        match e {                                            
       710 +            Expr::Fn(ps, b) | Expr::Macro(ps, b) => out.exten
           +d(escaping(ps, b)),                                          
       711 +            Expr::If(c, t, el) => {                          
       712 +                expr(c, out);                                
       713 +                stmts(t, out);                               
       714 +                if let Some(el) = el {                       
       715 +                    stmts(el, out);                          
       716 +                }                                            
       717 +            }                                                
       718 +            Expr::Prefix(_, r, _) => expr(r, out),           
       719 +            Expr::Infix(l, _, r, _) | Expr::Index(l, r, _) =>
           + {                                                           
       720 +                expr(l, out);                                
       721 +                expr(r, out);                                
       722 +            }                                                
       723 +            Expr::Call(c, args, _) => {                      
       724 +                expr(c, out);                                
       725 +                args.iter().for_each(|a| expr(a, out));      
       726 +            }                                                
       727 +            Expr::Array(xs) => xs.iter().for_each(|a| expr(a,
           + out)),                                                      
       728 +            Expr::Hash(ps) => ps.iter().for_each(|(k, v)| {  
       729 +                expr(k, out);                                
       730 +                expr(v, out)                                 
       731 +            }),                                              
       732 +            _ => {}                                          
       733 +        }                                                    
       734 +    }                                                        
       735 +    fn stmts(ss: &[Stmt], out: &mut HashSet<String>) {       
       736 +        for s in ss {                                        
       737 +            match s {                                        
       738 +                Stmt::Let(_, v, _) | Stmt::Return(v, _) | Stm
           +t::Expr(v) => expr(v, out),                                  
       739 +            }                                                
       740 +        }                                                    
       741 +    }                                                        
       742 +    let mut out = HashSet::new();                            
       743 +    stmts(body, &mut out);                                   
       744 +    out                                                      
       745 +}                                                            
       746 +                                                             
       747 +// ----------------------------------------------------------
           +-----------------                                            
       748 +// Code generation                                           
       749 +                                                             
       750 +/// One function environment being compiled.                 
       751 +struct Env {                                                 
       752 +    params: HashSet<String>,                                 
       753 +    lets: HashSet<String>,                                   
       754 +    decls: HashMap<String, DeclId>,                          
       755 +}                                                            
       756 +                                                             
       757 +struct Compiler {                                            
       758 +    g: Gen,                                                  
       759 +    /// Function environments, outermost first (the module to
           +p level is not                                               
       760 +    /// included: its bindings are globals).                 
       761 +    envs: Vec<Env>,                                          
       762 +    next_decl: DeclId,                                       
       763 +}                                                            
       764 +                                                             
       765 +impl Compiler {                                              
       766 +    fn line(&mut self, line: u32) {                          
       767 +        self.g.b().line = line;                              
       768 +    }                                                        
       769 +                                                             
       770 +    /// Emit code that pushes the value of identifier `name`.
       771 +    fn ident(&mut self, name: &str) {                        
       772 +        // Candidate environments, innermost first.          
       773 +        let mut cands: Vec<(DeclId, bool)> = Vec::new();     
       774 +        for env in self.envs.iter().rev() {                  
       775 +            if env.params.contains(name) {                   
       776 +                cands.push((env.decls[name], true));         
       777 +                break;                                       
       778 +            }                                                
       779 +            if env.lets.contains(name) {                     
       780 +                cands.push((env.decls[name], false));        
       781 +            }                                                
       782 +        }                                                    
       783 +        let definite = cands.last().is_some_and(|c| c.1);    
       784 +        if let [(d, true)] = cands[..] {                     
       785 +            let loc = self.g.resolve(d).expect("bound variabl
           +e");                                                         
       786 +            self.g.emit_get(loc);                            
       787 +            return;                                          
       788 +        }                                                    
       789 +        let end = self.g.b().new_label();                    
       790 +        for (i, &(d, def)) in cands.iter().enumerate() {     
       791 +            let loc = self.g.resolve(d).expect("bound variabl
           +e");                                                         
       792 +            self.g.emit_get(loc);                            
       793 +            if def {                                         
       794 +                debug_assert_eq!(i, cands.len() - 1);        
       795 +            } else {                                         
       796 +                self.g.b().jump(Op::JumpIfNotUndef, end);    
       797 +                self.g.emit(Op::Pop);                        
       798 +            }                                                
       799 +        }                                                    
       800 +        if !definite {                                       
       801 +            let k = self.g.pool.str(name);                   
       802 +            self.g.emit(Op::GetGlobalOpt(k));                
       803 +            self.g.b().jump(Op::JumpIfNotUndef, end);        
       804 +            self.g.emit(Op::Pop);                            
       805 +            match NativeId::lookup(Lang::Monkey, name) {     
       806 +                Some(id) => self.g.emit(Op::Native(id as u32)
           +),                                                           
       807 +                None => {                                    
       808 +                    let k = self.g.pool.str(&format!("identif
           +ier not found: {name}"));                                    
       809 +                    self.g.emit(Op::Fail(k));                
       810 +                    // Keep the static stack height consisten
           +t for the label.                                             
       811 +                    self.g.b().set_reachable(self.g.b().depth
           + + 1);                                                       
       812 +                    self.g.b().jump(Op::Jump, end);          
       813 +                }                                            
       814 +            }                                                
       815 +        }                                                    
       816 +        self.g.b().place(end);                               
       817 +    }                                                        
       818 +                                                             
       819 +    fn stmt(&mut self, s: &Stmt) {                           
       820 +        match s {                                            
       821 +            Stmt::Let(name, v, line) => {                    
       822 +                self.expr_named(v, Some(name));              
       823 +                self.line(*line);                            
       824 +                match self.envs.last() {

       200 +    Neg(Box<Exp>, Pos),                                      
       201 +    Record(String, Vec<(String, Exp, Pos)>, Pos),            
       202 +    Seq(Vec<Exp>),                                           
       203 +    Assign(Var, Box<Exp>, Pos),                              
       204 +    If(Box<Exp>, Box<Exp>, Option<Box<Exp>>, Pos),           
       205 +    While(Box<Exp>, Box<Exp>, Pos),                          
       206 +    For(String, Box<Exp>, Box<Exp>, Box<Exp>, Pos),          
       207 +    Break(Pos),                                              
       208 +    Let(Vec<Dec>, Vec<Exp>, Pos),                            
       209 +    Array(String, Box<Exp>, Box<Exp>, Pos),                  
       210 +}                                                            
       211 +                                                             
       212 +#[derive(Debug)]                                             
       213 +enum Var {                                                   
       214 +    Simple(String, Pos),                                     
       215 +    Field(Box<Var>, String, Pos),                            
       216 +    Subscript(Box<Var>, Box<Exp>, Pos),                      
       217 +}                                                            
       218 +                                                             
       219 +#[derive(Debug)]                                             
       220 +struct Field {                                               
       221 +    name: String,                                            
       222 +    typ: String,                                             
       223 +    pos: Pos,                                                
       224 +}                                                            
       225 +                                                             
       226 +#[derive(Debug)]                                             
       227 +struct FunDec {                                              
       228 +    name: String,                                            
       229 +    params: Vec<Field>,                                      
       230 +    result: Option<(String, Pos)>,                           
       231 +    /// `None` for a `primitive` declaration.                
       232 +    body: Option<Exp>,                                       
       233 +    pos: Pos,                                                
       234 +}                                                            
       235 +                                                             
       236 +#[derive(Debug)]                                             
       237 +enum Ty {                                                    
       238 +    Name(String, Pos),                                       
       239 +    Record(Vec<Field>),                                      
       240 +    Array(String, Pos),                                      
       241 +}                                                            
       242 +                                                             
       243 +#[derive(Debug)]                                             
       244 +enum Dec {                                                   
       245 +    Funs(Vec<FunDec>),                                       
       246 +    Var(String, Option<(String, Pos)>, Exp, Pos),            
       247 +    Types(Vec<(String, Ty, Pos)>),                           
       248 +}                                                            
       249 +                                                             
       250 +// ----------------------------------------------------------
           +-----------------                                            
       251 +// Parser                                                    
       252 +                                                             
       253 +struct Parser {                                              
       254 +    toks: Vec<(Tok, Pos)>,                                   
       255 +    pos: usize,                                              
       256 +}                                                            
       257 +                                                             
       258 +type PR<T> = Result<T, (Pos, String)>;                       
       259 +                                                             
       260 +impl Parser {                                                
       261 +    fn peek(&self) -> &Tok {                                 
       262 +        &self.toks[self.pos].0                               
       263 +    }                                                        
       264 +    fn peek2(&self) -> &Tok {                                
       265 +        &self.toks[(self.pos + 1).min(self.toks.len() - 1)].0
       266 +    }                                                        
       267 +    fn here(&self) -> Pos {                                  
       268 +        self.toks[self.pos].1                                
       269 +    }                                                        
       270 +    fn advance(&mut self) -> Tok {                           
       271 +        let t = self.toks[self.pos].0.clone();               
       272 +        if self.pos < self.toks.len() - 1 {                  
       273 +            self.pos += 1;                                   
       274 +        }                                                    
       275 +        t                                                    
       276 +    }                                                        
       277 +    fn is(&self, t: &Tok) -> bool {                          
       278 +        self.peek() == t                                     
       279 +    }                                                        
       280 +    fn is_p(&self, p: &str) -> bool {                        
       281 +        matches!(self.peek(), Tok::P(q) if *q == p)          
       282 +    }                                                        
       283 +    fn is_kw(&self, k: &str) -> bool {                       
       284 +        matches!(self.peek(), Tok::Kw(q) if *q == k)         
       285 +    }                                                        
       286 +    fn err<T>(&self, expected: &str) -> PR<T> {              
       287 +        Err((self.here(), format!("syntax error: expected {ex
           +pected}, found {}", self.peek().describe())))                
       288 +    }                                                        
       289 +    fn expect_p(&mut self, p: &'static str) -> PR<()> {      
       290 +        if self.is_p(p) {                                    
       291 +            self.advance();                                  
       292 +            Ok(())                                           
       293 +        } else {                                             
       294 +            self.err(&format!("'{p}'"))                      
       295 +        }                                                    
       296 +    }                                                        
       297 +    fn expect_kw(&mut self, k: &'static str) -> PR<()> {     
       298 +        if self.is_kw(k) {                                   
       299 +            self.advance();                                  
       300 +            Ok(())                                           
       301 +        } else {                                             
       302 +            self.err(&format!("'{k}'"))                      
       303 +        }                                                    
       304 +    }                                                        
       305 +    fn ident(&mut self) -> PR<(String, Pos)> {               
       306 +        let pos = self.here();                               
       307 +        match self.peek().clone() {                          
       308 +            Tok::Id(s) => {                                  
       309 +                self.advance();                              
       310 +                Ok((s, pos))                                 
       311 +            }                                                
       312 +            _ => self.err("an identifier"),                  
       313 +        }                                                    
       314 +    }                                                        
       315 +                                                             
       316 +    fn program(&mut self) -> PR<Exp> {                       
       317 +        let e = self.exp()?;                                 
       318 +        if !self.is(&Tok::Eof) {                             
       319 +            return self.err("end of file");                  
       320 +        }                                                    
       321 +        Ok(e)                                                
       322 +    }                                                        
       323 +                                                             
       324 +    fn exp(&mut self) -> PR<Exp> {                           
       325 +        let pos = self.here();                               
       326 +        let e = self.or_exp()?;                              
       327 +        if self.is_p(":=") {                                 
       328 +            let apos = self.here();                          
       329 +            let Exp::Var(v) = e else {                       
       330 +                return Err((pos, "syntax error: the left side
           + of ':=' must be a variable, field or subscript".into()));   
       331 +            };                                               
       332 +            self.advance();                                  
       333 +            let rhs = self.exp()?;                           
       334 +            return Ok(Exp::Assign(v, Box::new(rhs), apos));  
       335 +        }                                                    
       336 +        Ok(e)                                                
       337 +    }                                                        
       338 +                                                             
       339 +    fn or_exp(&mut self) -> PR<Exp> {                        
       340 +        let mut e = self.and_exp()?;                         
       341 +        while self.is_p("|") {                               
       342 +            let pos = self.here();                           
       343 +            self.advance();                                  
       344 +            let r = self.and_exp()?;                         
       345 +            e = Exp::Op(Box::new(e), "|", Box::new(r), pos); 
       346 +        }                                                    
       347 +        Ok(e)                                                
       348 +    }                                                        
       349 +                                                             
       350 +    fn and_exp(&mut self) -> PR<Exp> {                       
       351 +        let mut e = self.cmp_exp()?;                         
       352 +        while self.is_p("&") {                               
       353 +            let pos = self.here();                           
       354 +            self.advance();                                  
       355 +            let r = self.cmp_exp()?;                         
       356 +            e = Exp::Op(Box::new(e), "&", Box::new(r), pos); 
       357 +        }                                                    
       358 +        Ok(e)                                                
       359 +    }                                                        
       360 +                                                             
       361 +    fn cmp_op(&self) -> Option<&'static str> {               
       362 +        match self.peek() {                                  
       363 +            Tok::P(p) if ["=", "<>", "<", "<=", ">", ">="].co
           +ntains(p) => Some(p),                                        
       364 +            _ => None,                                       
       365 +        }                                                    
       366 +    }                                                        
       367 +                                                             
       368 +    fn cmp_exp(&mut self) -> PR<Exp> {                       
       369 +        let e = self.add_exp()?;                             
       370 +        if let Some(op) = self.cmp_op() {                    
       371 +            let pos = self.here();                           
       372 +            self.advance();                                  
       373 +            let r = self.add_exp()?;                         
       374 +            if self.cmp_op().is_some() {                     
       375 +                return Err((self.here(), "syntax error: compa
           +rison operators do not associate".into()));                  
       376 +            }                                                
       377 +            return Ok(Exp::Op(Box::new(e), op, Box::new(r), p
           +os));                                                        
       378 +        }                                                    
       379 +        Ok(e)                                                
       380 +    }                                                        
       381 +                                                             
       382 +    fn add_exp(&mut self) -> PR<Exp> {                       
       383 +        let mut e = self.mul_exp()?;                         
       384 +        while self.is_p("+") || self.is_p("-") {             
       385 +            let pos = self.here();                           
       386 +            let Tok::P(op) = self.advance() else { unreachabl
           +e!() };                                                      
       387 +            let r = self.mul_exp()?;                         
       388 +            e = Exp::Op(Box::new(e), op, Box::new(r), pos);  
       389 +        }                                                    
       390 +        Ok(e)                                                
       391 +    }                                                        
       392 +                                                             
       393 +    fn mul_exp(&mut self) -> PR<Exp> {                       
       394 +        let mut e = self.unary()?;                           
       395 +        while self.is_p("*") || self.is_p("/") {             
       396 +            let pos = self.here();                           
       397 +            let Tok::P(op) = self.advance() else { unreachabl
           +e!() };                                                      
       398 +            let r = self.unary()?;                           
       399 +            e = Exp::Op(Box::new(e), op, Box::new(r), pos);  
       400 +        }                                                    
       401 +        Ok(e)                                                
       402 +    }                                                        
       403 +                                                             
       404 +    fn unary(&mut self) -> PR<Exp> {                         
       405 +        if self.is_p("-") {                                  
       406 +            let pos = self.here();                           
       407 +            self.advance();                                  
       408 +            let e = self.unary()?;                           
       409 +            return Ok(Exp::Neg(Box::new(e), pos));           
       410 +        }                                                    
       411 +        self.primary()                                       
       412 +    }                                                        
       413 +                                                             
       414 +    fn seq(&mut self, end: &Tok) -> PR<Vec<Exp>> {           
       415 +        let mut es = Vec::new();                             
       416 +        if self.is(end) {                                    
       417 +            return Ok(es);                                   
       418 +        }                                                    
       419 +        es.push(self.exp()?);                                
       420 +        while self.is_p(";") {                               
       421 +            self.advance();                                  
       422 +            es.push(self.exp()?);                            
       423 +        }                                                    
       424 +        Ok(es)                                               
       425 +    }                                                        
       426 +                                                             
       427 +    fn primary(&mut self) -> PR<Exp> {                       
       428 +        let pos = self.here();                               
       429 +        match self.peek().clone() {                          
       430 +            Tok::Int(i) => {                                 
       431 +                self.advance();                              
       432 +                Ok(Exp::Int(i))                              
       433 +            }                                                
       434 +            Tok::Str(s) => {                                 
       435 +                self.advance();                              
       436 +                Ok(Exp::Str(s))                              
       437 +            }                                                
       438 +            Tok::Kw("nil") => {                              
       439 +                self.advance();                              
       440 +                Ok(Exp::Nil)                                 
       441 +            }                                                
       442 +            Tok::Kw("break") => {                            
       443 +                self.advance();                              
       444 +                Ok(Exp::Break(pos))                          
       445 +            }                                                
       446 +            Tok::P("(") => {                                 
       447 +                self.advance();                              
       448 +                let es = self.seq(&Tok::P(")"))?;            
       449 +                self.expect_p(")")?;                         
       450 +                Ok(Exp::Seq(es))                             
       451 +            }                                                
       452 +            Tok::Kw("if") => {                               
       453 +                self.advance();                              
       454 +                let c = self.exp()?;                         
       455 +                self.expect_kw("then")?;                     
       456 +                let t = self.exp()?;                         
       457 +                let e = if self.is_kw("else") {              
       458 +                    self.advance();                          
       459 +                    Some(Box::new(self.exp()?))              
       460 +                } else {                                     
       461 +                    None                                     
       462 +                };                                           
       463 +                Ok(Exp::If(Box::new(c), Box::new(t), e, pos))
       464 +            }                                                
       465 +            Tok::Kw("while") => {                            
       466 +                self.advance();                              
       467 +                let c = self.exp()?;                         
       468 +                self.expect_kw("do")?;                       
       469 +                let b = self.exp()?;                         
       470 +                Ok(Exp::While(Box::new(c), Box::new(b), pos))
       471 +            }                                                
       472 +            Tok::Kw("for") => {                              
       473 +                self.advance();                              
       474 +                let (v, _) = self.ident()?;                  
       475 +                self.expect_p(":=")?;                        
       476 +                let lo = self.exp()?;                        
       477 +                self.expect_kw("to")?;                       
       478 +                let hi = self.exp()?;                        
       479 +                self.expect_kw("do")?;                       
       480 +                let b = self.exp()?;                         
       481 +                Ok(Exp::For(v, Box::new(lo), Box::new(hi), Bo
           +x::new(b), pos))                                             
       482 +            }                                                
       483 +            Tok::Kw("let") => {                              
       484 +                self.advance();                              
       485 +                let decs = self.decs()?;                     
       486 +                self.expect_kw("in")?;                       
       487 +                let body = self.seq(&Tok::Kw("end"))?;       
       488 +                self.expect_kw("end")?;                      
       489 +                Ok(Exp::Let(decs, body, pos))                
       490 +            }                                                
       491 +            Tok::Id(name) => {                               
       492 +                self.advance();                              
       493 +                if self.is_p("(") {                          
       494 +                    self.advance();                          
       495 +                    let mut args = Vec::new();               
       496 +                    if !self.is_p(")") {                     
       497 +                        args.push(self.exp()?);              
       498 +                        while self.is_p(",") {               
       499 +                            self.advance();                  
       500 +                            args.push(self.exp()?);          
       501 +                        }                                    
       502 +                    }                                        
       503 +                    self.expect_p(")")?;                     
       504 +                    return Ok(Exp::Call(name, args, pos));   
       505 +                }                                            
       506 +                if self.is_p("{") {                          
       507 +                    self.advance();                          
       508 +                    let mut fields = Vec::new();             
       509 +                    if !self.is_p("}") {                     
       510 +                        loop {                               
       511 +                            let (f, fpos) = self.ident()?;   
       512 +                            self.expect_p("=")?;             
       513 +                            let e = self.exp()?;             
       514 +                            fields.push((f, e, fpos));       
       515 +                            if !self.is_p(",") {             
       516 +                                break;                       
       517 +                            }                                
       518 +                            self.advance();                  
       519 +                        }                                    
       520 +                    }                                        
       521 +                    self.expect_p("}")?;                     
       522 +                    return Ok(Exp::Record(name, fields, pos))
           +;                                                            
       523 +                }                                            
       524 +                let mut v = Var::Simple(name.clone(), pos);  
       525 +                if self.is_p("[") {                          
       526 +                    let bpos = self.here();                  
       527 +                    self.advance();                          
       528 +                    let idx = self.exp()?;                   
       529 +                    self.expect_p("]")?;                     
       530 +                    if self.is_kw("of") {                    
       531 +                        self.advance();                      
       532 +                        let init = self.exp()?;              
       533 +                        return Ok(Exp::Array(name, Box::new(i
           +dx), Box::new(init), pos));                                  
       534 +                    }                                        
       535 +                    v = Var::Subscript(Box::new(v), Box::new(
           +idx), bpos);                                                 
       536 +                }                                            
       537 +                loop {                                       
       538 +                    let p = self.here();                     
       539 +                    if self.is_p(".") {                      
       540 +                        self.advance();                      
       541 +                        let (f, _) = self.ident()?;          
       542 +                        v = Var::Field(Box::new(v), f, p);   
       543 +                    } else if self.is_p("[") {               
       544 +                        self.advance();                      
       545 +                        let idx = self.exp()?;               
       546 +                        self.expect_p("]")?;                 
       547 +                        v = Var::Subscript(Box::new(v), Box::
           +new(idx), p);                                                
       548 +                    } else {                                 
       549 +                        return Ok(Exp::Var(v));              
       550 +                    }                                        
       551 +                }                                            
       552 +            }                                                
       553 +            _ => self.err("an expression"),                  
       554 +        }                                                    
       555 +    }                                                        
       556 +                                                             
       557 +    fn type_fields(&mut self, end: &str) -> PR<Vec<Field>> { 
       558 +        let mut fs = Vec::new();                             
       559 +        if self.is_p(end) {                                  
       560 +            return Ok(fs);                                   
       561 +        }                                                    
       562 +        loop {                                               
       563 +            let (name, pos) = self.ident()?;                 
       564 +            self.expect_p(":")?;                             
       565 +            let (typ, _) = self.ident()?;                    
       566 +            fs.push(Field { name, typ, pos });               
       567 +            if !self.is_p(",") {                             
       568 +                return Ok(fs);                               
       569 +            }                                                
       570 +            self.advance();                                  
       571 +        }                                                    
       572 +    }                                                        
       573 +                                                             
       574 +    fn is_primitive(&self) -> bool {                         
       575 +        matches!(self.peek(), Tok::Id(s) if s == "primitive")
           + && matches!(self.peek2(), Tok::Id(_))                       
       576 +    }                                                        
       577 +                                                             
       578 +    fn decs(&mut self) -> PR<Vec<Dec>> {                     
       579 +        let mut decs: Vec<Dec> = Vec::new();                 
       580 +        loop {                                               
       581 +            let pos = self.here();                           
       582 +            if self.is_kw("type") {                          
       583 +                self.advance();                              
       584 +                let (name, _) = self.ident()?;               
       585 +                self.expect_p("=")?;                         
       586 +                let tpos = self.here();                      
       587 +                let ty = if self.is_p("{") {                 
       588 +                    self.advance();                          
       589 +                    let fs = self.type_fields("}")?;         
       590 +                    self.expect_p("}")?;                     
       591 +                    Ty::Record(fs)                           
       592 +                } else if self.is_kw("array") {              
       593 +                    self.advance();                          
       594 +                    self.expect_kw("of")?;                   
       595 +                    let (t, p) = self.ident()?;              
       596 +                    Ty::Array(t, p)                          
       597 +                } else {                                     
       598 +                    let (t, _) = self.ident()?;              
       599 +                    Ty::Name(t, tpos)                        
       600 +                };                                           
       601 +                match decs.last_mut() {                      
       602 +                    Some(Dec::Types(g)) => g.push((name, ty, 
           +pos)),                                                       
       603 +                    _ => decs.push(Dec::Types(vec![(name, ty,
           + pos)])),                                                    
       604 +                }                                            
       605 +            } else if self.is_kw("var") {                    
       606 +                self.advance();                              
       607 +                let (name, _) = self.ident()?;               
       608 +                let typ = if self.is_p(":") {                
       609 +                    self.advance();                          
       610 +                    Some(self.ident()?)                      
       611 +                } else {                                     
       612 +                    None                                     
       613 +                };                                           
       614 +                self.expect_p(":=")?;                        
       615 +                let init = self.exp()?;                      
       616 +                decs.push(Dec::Var(name, typ, init, pos));   
       617 +            } else if self.is_kw("function") || self.is_primi
           +tive() {                                                     
       618 +                let primitive = !self.is_kw("function");     
       619 +                self.advance();                              
       620 +                let (name, _) = self.ident()?;               
       621 +                self.expect_p("(")?;                         
       622 +                let params = self.type_fields(")")?;         
       623 +                self.expect_p(")")?;                         
       624 +                let result = if self.is_p(":") {             
       625 +                    self.advance();                          
       626 +                    Some(self.ident()?)                      
       627 +                } else {                                     
       628 +                    None                                     
       629 +                };                                           
       630 +                let body = if primitive {                    
       631 +                    None                                     
       632 +                } else {                                     
       633 +                    self.expect_p("=")?;                     
       634 +                    Some(self.exp()?)                        
       635 +                };                                           
       636 +                let f = FunDec { name, params, result, body, 
           +pos };                                                       
       637 +                match decs.last_mut() {                      
       638 +                    Some(Dec::Funs(g)) => g.push(f),         
       639 +                    _ => decs.push(Dec::Funs(vec![f])),      
       640 +                }                                            
       641 +            } else {                                         
       642 +                return Ok(decs);                             
       643 +            }                                                
       644 +        }                                                    
       645 +    }                                                        
       646 +}                                                            
       647 +                                                             
       648 +// ----------------------------------------------------------
           +-----------------                                            
       649 +// Types and the typed tree produced by the checker          
       650 +                                                             
       651 +#[derive(Clone, Copy, Debug, PartialEq, Eq)]                 
       652 +enum Type {                                                  
       653 +    Int,                                                     
       654 +    Str,                                                     
       655 +    Nil,                                                     
       656 +    Unit,                                                    
       657 +    Record(usize),                                           
       658 +    Array(usize),                                            
       659 +    /// A name from a type declaration, resolved through `Sem
           +ant::names`.                                                 
       660 +    Name(usize),                                             
       661 +    /// Result of an erroneous expression; compatible with ev
           +erything so                                                  
       662 +    /// that one mistake produces one message.               
       663 +    Error,                                                   
       664 +}                                                            
       665 +                                                             
       666 +struct RecordInfo {                                          
       667 +    name: String,                                            
       668 +    fields: Vec<(String, Type)>,                             
       669 +}                                                            
       670 +                                                             
       671 +struct ArrayInfo {                                           
       672 +    name: String,                                            
       673 +    elem: Type,                                              
       674 +}                                                            
       675 +                                                             
       676 +#[derive(Clone, Copy, Debug)]                                
       677 +enum CheckKind {                                             
       678 +    Int,                                                     
       679 +    Str,                                                     
       680 +    None,                                                    
       681 +}                                                            
       682 +                                                             
       683 +#[derive(Debug)]                                             
       684 +enum Callee {                                                
       685 +    Local(DeclId),                                           
       686 +    Native(NativeId),                                        
       687 +    Extern(String, CheckKind),                               
       688 +}                                                            
       689 +                                                             
       690 +/// A typed expression.  `value` says whether its code leaves
           + a value on the                                              
       691 +/// stack (expressions of type unit leave nothing).          
       692 +#[derive(Debug)]                                             
       693 +struct TE {                                                  
       694 +    k: TK,                                                   
       695 +    value: bool,                                             
       696 +    line: u32,                                               
       697 +}                                                            
       698 +                                                             
       699 +#[derive(Debug)]                                             
       700 +enum TK {                                                    
       701 +    Int(i64),                                                
       702 +    Str(String),                                             
       703 +    Nil,                                                     
       704 +    Get(DeclId),                                             
       705 +    Set(DeclId, Box<TE>),                                    
       706 +    Field(Box<TE>, u32),                                     
       707 +    SetField(Box<TE>, u32, Box<TE>),                         
       708 +    Sub(Box<TE>, Box<TE>),                                   
       709 +    SetSub(Box<TE>, Box<TE>, Box<TE>),                       
       710 +    Call(Callee, Vec<TE>),                                   
       711 +    Bin(Op, Box<TE>, Box<TE>),                               
       712 +    /// Comparison producing 0/1 (the op yields a boolean).  
       713 +    Cmp(Op, Box<TE>, Box<TE>),                               
       714 +    And(Box<TE>, Box<TE>),                                   
       715 +    Or(Box<TE>, Box<TE>),                                    
       716 +    Neg(Box<TE>),                                            
       717 +    Record(usize, Vec<TE>),                                  
       718 +    Array(Box<TE>, Box<TE>),                                 
       719 +    Seq(Vec<TE>),                                            
       720 +    If(Box<TE>, Box<TE>, Option<Box<TE>>),                   
       721 +    While(Box<TE>, Box<TE>),                                 
       722 +    For(DeclId, Box<TE>, Box<TE>, Box<TE>),                  
       723 +    Break,                                                   
       724 +    Let(Vec<TD>, Box<TE>),                                   
       725 +}                                                            
       726 +                                                             
       727 +#[derive(Debug)]                                             
       728 +enum TD {                                                    
       729 +    Var(DeclId, TE),                                         
       730 +    Funs(Vec<TFun>),                                         
       731 +}                                                            
       732 +                                                             
       733 +#[derive(Debug)]                                             
       734 +struct TFun {                                                
       735 +    decl: DeclId,                                            
       736 +    name: String,                                            
       737 +    params: Vec<(DeclId, CheckKind)>,                        
       738 +    body: TE,                                                
       739 +    returns_value: bool,                                     
       740 +    exported: bool,                                          
       741 +}                                                            
       742 +                                                             
       743 +#[derive(Clone)]                                             
       744 +enum VEntry {                                                
       745 +    Var { decl: DeclId, ty: Type, readonly: bool },          
       746 +    Fun { callee: FunRef, formals: Vec<Type>, result: Type },
       747 +}                                                            
       748 +                                                             
       749 +#[derive(Clone)]                                             
       750 +enum FunRef {                                                
       751 +    Local(DeclId),                                           
       752 +    Native(NativeId),                                        
       753 +    Extern(String),                                          
       754 +}                                                            
       755 +                                                             
       756 +/// A symbol table with nested scopes.                       
       757 +struct Scoped<V> {                                           
       758 +    scopes: Vec<HashMap<String, V>>,                         
       759 +}                                                            
       760 +                                                             
       761 +impl<V: Clone> Scoped<V> {                                   
       762 +    fn new() -> Self {                                       
       763 +        Scoped { scopes: vec![HashMap::new()] }              
       764 +    }                                                        
       765 +    fn push(&mut self) {                                     
       766 +        self.scopes.push(HashMap::new())                     
       767 +    }                                                        
       768 +    fn pop(&mut self) {                                      
       769 +        self.scopes.pop();                                   
       770 +    }                                                        
       771 +    fn insert(&mut self, k: &str, v: V) {                    
       772 +        self.scopes.last_mut().unwrap().insert(k.to_string(),
           + v);                                                         
       773 +    }                                                        
       774 +    fn get(&self, k: &str) -> Option<V> {                    
       775 +        self.scopes.iter().rev().find_map(|s| s.get(k).cloned
           +())                                                          
       776 +    }                                                        
       777 +}                                                            
       778 +                                                             
       779 +// ----------------------------------------------------------
           +-----------------                                            
       780 +// Type checker                                              
       781 +                                                             
       782 +struct Semant {                                              
       783 +    tenv: Scoped<Type>,                                      
       784 +    venv: Scoped<VEntry>,                                    
       785 +    records: Vec<RecordInfo>,                                
       786 +    arrays: Vec<ArrayInfo>,                                  
       787 +    names: Vec<(String, Option<Type>)>,                      
       788 +    errors: Vec<String>,                                     
       789 +    file: String,                                            
       790 +    /// Function currently being checked (0 = the main progra
           +m).                                                          
       791 +    cur_fn: usize,                                           
       792 +    next_fn: usize,                                          
       793 +    decl_owner: Vec<usize>,                                  
       794 +    captured: HashSet<DeclId>,                               
       795 +    loop_depth: usize,                                       
       796 +}                                                            
       797 +                                                             
       798 +fn check_kind(t: Type) -> CheckKind {                        
       799 +    match t {                                                
       800 +        Type::Int => CheckKind::Int,                         
       801 +        Type::Str => CheckKind::Str,                         
       802 +        _ => CheckKind::None,                                
       803 +    }                                                        
       804 +}                                                            
       805 +                                                             
       806 +impl Semant {                                                
       807 +    fn new(file: &str) -> Semant {                           
       808 +        let mut s = Semant {                                 
       809 +            tenv: Scoped::new(),                             
       810 +            venv: Scoped::new(),                             
       811 +            records: Vec::new(),                             
       812 +            arrays: Vec::new(),                              
       813 +            names: Vec::new(),                               
       814 +            errors: Vec::new(),                              
       815 +            file: file.to_string(),                          
       816 +            cur_fn: 0,                                       
       817 +            next_fn: 1,                                      
       818 +            decl_owner: Vec::new(),                          
       819 +            captured: HashSet::new(),                        
       820 +            loop_depth: 0,                                   
       821 +        };                                                   
       822 +        s.tenv.insert("int", Type::Int);                     
       823 +        s.tenv.insert("string", Type::Str);                  
       824 +        use Type::{Int, Str, Unit};                          
       825 +        let lib: &[(&str, &[Type], Type)] = &[               
       826 +            ("print", &[Str], Unit),                         
       827 +            ("printi", &[Int], Unit),                        
       828 +            ("flush", &[], Unit),                            
       829 +            ("getchar", &[], Str),                           
       830 +            ("ord", &[Str], Int),                            
       831 +            ("chr", &[Int], Str),                            
       832 +            ("size", &[Str], Int),                           
       833 +            ("substring", &[Str, Int, Int], Str),            
       834 +            ("concat", &[Str, Str], Str),                    
       835 +            ("not", &[Int], Int),                            
       836 +            ("exit", &[Int], Unit),                          
       837 +        ];                                                   
       838 +        for (name, formals, result) in lib {                 
       839 +            let id = NativeId::lookup(Lang::Tiger, name).expe
           +ct("Tiger native");                                          
       840 +            s.venv.insert(name, VEntry::Fun { callee: FunRef:
           +:Native(id), formals: formals.to_vec(), result: *result });  
       841 +        }                                                    
       842 +        s                                                    
       843 +    }                                                        
       844 +                                                             
       845 +    fn error(&mut self, pos: Pos, msg: impl Into<String>) {  
       846 +        self.errors.push(format!("{}:{}:{}: error: {}", self.
           +file, pos.0, pos.1, msg.into()));                            
       847 +    }                                                        
       848 +                                                             
       849 +    fn new_decl(&mut self) -> DeclId {                       
       850 +        self.decl_owner.push(self.cur_fn);                   
       851 +        self.decl_owner.len() - 1                            
       852 +    }                                                        
       853 +                                                             
       854 +    fn use_decl(&mut self, d: DeclId) {                      
       855 +        if self.decl_owner[d] != self.cur_fn {               
       856 +            self.captured.insert(d);                         
       857 +        }                                                    
       858 +    }                                                        
       859 +                                                             
       860 +    /// Follow type-name links to the underlying type.       
       861 +    fn actual(&self, mut t: Type) -> Type {                  
       862 +        for _ in 0..=self.names.len() {                      
       863 +            match t {                                        
       864 +                Type::Name(n) => match self.names[n].1 {     
       865 +                    Some(u) => t = u,                        
       866 +                    None => return Type::Error,              
       867 +                },                                           
       868 +                other => return other,                       
       869 +            }                                                
       870 +        }                                                    
       871 +        Type::Error // cycle; reported at declaration        
       872 +    }                                                        
       873 +                                                             
       874 +    fn type_name(&self, t: Type) -> String {                 
       875 +        match self.actual(t) {                               
       876 +            Type::Int => "int".into(),                       
       877 +            Type::Str => "string".into(),                    
       878 +            Type::Nil => "nil".into(),                       
       879 +            Type::Unit => "no value".into(),                 
       880 +            Type::Record(r) => format!("record type '{}'", se
           +lf.records[r].name),                                         
       881 +            Type::Array(a) => format!("array type '{}'", self
           +.arrays[a].name),                                            
       882 +            Type::Name(_) | Type::Error => "<error>".into(), 
       883 +        }                                                    
       884 +    }                                                        
       885 +                                                             
       886 +    /// Can a value of type `actual` be used where `expected`
           + is required?                                                
       887 +    fn fits(&self, actual: Type, expected: Type) -> bool {   
       888 +        let (a, e) = (self.actual(actual), self.actual(expect
           +ed));                                                        
       889 +        a == e || a == Type::Error || e == Type::Error || (a 
           +== Type::Nil && matches!(e, Type::Record(_)))                
       890 +    }                                                        
       891 +                                                             
       892 +    fn lookup_type(&mut self, name: &str, pos: Pos) -> Type {
       893 +        match self.tenv.get(name) {                          
       894 +            Some(t) => t,                                    
       895 +            None => {                                        
       896 +                self.error(pos, format!("undefined type '{nam
           +e}'"));                                                      
       897 +                Type::Error                                  
       898 +            }                                                
       899 +        }                                                    
       900 +    }                                                        
       901 +                                                             
       902 +    fn te(k: TK, ty: Type, line: u32) -> (TE, Type) {        
       903 +        (TE { k, value: ty != Type::Unit, line }, ty)        
       904 +    }                                                        
       905 +                                                             
       906 +    fn check_int(&mut self, ty: Type, pos: Pos, what: &str) {
       907 +        if !self.fits(ty, Type::Int) {                       
       908 +            let found = self.type_name(ty);                  
       909 +            self.error(pos, format!("{what} must be an intege
           +r, found {found}"));                                         
       910 +        }                                                    
       911 +    }                                                        
       912 +                                                             
       913 +    fn exp(&mut self, e: &Exp, line: u32) -> (TE, Type) {    
       914 +        match e {                                            
       915 +            Exp::Nil => Self::te(TK::Nil, Type::Nil, line),  
       916 +            Exp::Int(i) => Self::te(TK::Int(*i), Type::Int, l
           +ine),                                                        
       917 +            Exp::Str(s) => Self::te(TK::Str(s.clone()), Type:
           +:Str, line),                                                 
       918 +            Exp::Var(v) => self.var(v),                      
       919 +            Exp::Call(name, args, pos) => {                  
       920 +                let targs: Vec<(TE, Type, Pos)> = args       
       921 +                    .iter()                                  
       922 +                    .map(|a| {                               
       923 +                        let (t, ty) = self.exp(a, pos.0);    
       924 +                        (t, ty, *pos)                        
       925 +                    })                                       
       926 +                    .collect();                              
       927 +                match self.venv.get(name) {                  
       928 +                    Some(VEntry::Fun { callee, formals, resul
           +t }) => {                                                    
       929 +                        if formals.len() != targs.len() {    
       930 +                            self.error(                      
       931 +                                *pos,                        
       932 +                                format!("function '{name}' ex
           +pects {} arguments but got {}", formals.len(), targs.len()), 
       933 +                            );                               
       934 +                        } else {                             
       935 +                            for (i, ((_, ty, _), f)) in targs
           +.iter().zip(&formals).enumerate() {                          
       936 +                                if !self.fits(*ty, *f) {     
       937 +                                    let (want, got) = (self.t
           +ype_name(*f), self.type_name(*ty));                          
       938 +                                    self.error(*pos, format!(
           +"argument {} of '{name}' must be {want}, found {got}", i + 1)
           +);                                                           
       939 +                                }                            
       940 +                            }                                
       941 +                        }                                    
       942 +                        let callee = match callee {          
       943 +                            FunRef::Local(d) => {            
       944 +                                self.use_decl(d);            
       945 +                                Callee::Local(d)             
       946 +                            }                                
       947 +                            FunRef::Native(id) => Callee::Nat
           +ive(id),                                                     
       948 +                            FunRef::Extern(n) => Callee::Exte
           +rn(n, check_kind(self.actual(result))),                      
       949 +                        };                                   
       950 +                        let args = targs.into_iter().map(|a| 
           +a.0).collect();                                              
       951 +                        Self::te(TK::Call(callee, args), resu
           +lt, pos.0)                                                   
       952 +                    }                                        
       953 +                    Some(VEntry::Var { .. }) => {            
       954 +                        self.error(*pos, format!("'{name}' is
           + a variable, not a function"));                              
       955 +                        Self::te(TK::Nil, Type::Error, pos.0)
       956 +                    }                                        
       957 +                    None => {                                
       958 +                        self.error(*pos, format!("undefined f
           +unction '{name}'"));                                         
       959 +                        Self::te(TK::Nil, Type::Error, pos.0)
       960 +                    }                                        
       961 +                }                                            
       962 +            }                                                
       963 +            Exp::Op(a, op, b, pos) => {                      
       964 +                let (ta, tya) = self.exp(a, pos.0);          
       965 +                let (tb, tyb) = self.exp(b, pos.0);          
       966 +                let (a, b) = (Box::new(ta), Box::new(tb));   
       967 +                match *op {                                  
       968 +                    "+" | "-" | "*" | "/" | "&" | "|" => {   
       969 +                        self.check_int(tya, *pos, &format!("l
           +eft operand of '{op}'"));                                    
       970 +                        self.check_int(tyb, *pos, &format!("r
           +ight operand of '{op}'"));                                   
       971 +                        let k = match *op {                  
       972 +                            "+" => TK::Bin(Op::Add, a, b),   
       973 +                            "-" => TK::Bin(Op::Sub, a, b),   
       974 +                            "*" => TK::Bin(Op::Mul, a, b),   
       975 +                            "/" => TK::Bin(Op::Div, a, b),   
       976 +                            "&" => TK::And(a, b),            
       977 +                            _ => TK::Or(a, b),               
       978 +                        };                                   
       979 +                        Self::te(k, Type::Int, pos.0)        
       980 +                    }                                        
       981 +                    "<" | "<=" | ">" | ">=" => {             
       982 +                        let (x, y) = (self.actual(tya), self.
           +actual(tyb));                                                
       983 +                        let strings = match (x, y) {         
       984 +                            (Type::Int | Type::Error, Type::I
           +nt | Type::Error) => false,                                  
       985 +                            (Type::Str | Type::Error, Type::S
           +tr | Type::Error) => true,                                   
       986 +                            _ => {                           
       987 +                                let (n1, n2) = (self.type_nam
           +e(x), self.type_name(y));                                    
       988 +                                self.error(*pos, format!("can
           +not compare {n1} with {n2} using '{op}' (both operands must b
           +e integers or both strings)"));                              
       989 +                                false                        
       990 +                            }                                
       991 +                        };                                   
       992 +                        let o = match (*op, strings) {       
       993 +                            ("<", false) => Op::Lt,          
       994 +                            ("<=", false) => Op::Le,         
       995 +                            (">", false) => Op::Gt,          
       996 +                            (">=", false) => Op::Ge,         
       997 +                            ("<", true) => Op::SLt,          
       998 +                            ("<=", true) => Op::SLe,         
       999 +                            (">", true) => Op::SGt,          
      1000 +                            _ => Op::SGe,                    
      1001 +                        };                                   
      1002 +                        Self::te(TK::Cmp(o, a, b), Type::Int,
           + pos.0)                                                      
      1003 +                    }                                        
      1004 +                    _ => {                                   
      1005 +                        let (x, y) = (self.actual(tya), self.
           +actual(tyb));                                                
      1006 +                        let ok = match (x, y) {              
      1007 +                            (Type::Error, _) | (_, Type::Erro
           +r) => true,                                                  
      1008 +                            (Type::Unit, _) | (_, Type::Unit)
           + => false,                                                   
      1009 +                            (Type::Nil, Type::Nil) => false, 
      1010 +                            (Type::Nil, Type::Record(_)) | (T
           +ype::Record(_), Type::Nil) => true,                          
      1011 +                            _ => x == y,                     
      1012 +                        };                                   
      1013 +                        if !ok {                             
      1014 +                            let msg = if x == Type::Nil && y 
           +== Type::Nil {                                               
      1015 +                                "cannot compare nil with nil:
           + the record type is unknown".to_string()                     
      1016 +                            } else {                         
      1017 +                                format!("cannot compare {} wi
           +th {} using '{op}'", self.type_name(x), self.type_name(y))   
      1018 +                            };                               
      1019 +                            self.error(*pos, msg);           
      1020 +                        }                                    
      1021 +                        Self::te(TK::Cmp(if *op == "=" { Op::
           +Eq } else { Op::Ne }, a, b), Type::Int, pos.0)               
      1022 +                    }                                        
      1023 +                }                                            
      1024 +            }                                                
      1025 +            Exp::Neg(a, pos) => {                            
      1026 +                let (t, ty) = self.exp(a, pos.0);            
      1027 +                self.check_int(ty, *pos, "operand of unary '-
           +'");                                                         
      1028 +                Self::te(TK::Neg(Box::new(t)), Type::Int, pos
           +.0)                                                          
      1029 +            }                                                
      1030 +            Exp::Record(tname, fields, pos) => {             
      1031 +                let t = self.lookup_type(tname, *pos);       
      1032 +                let rt = self.actual(t);                     
      1033 +                let tfields: Vec<(TE, Type)> = fields.iter().
           +map(|(_, e, p)| self.exp(e, p.0)).collect();                 
      1034 +                match rt {                                   
      1035 +                    Type::Record(r) => {                     
      1036 +                        let decl: Vec<(String, Type)> = self.
           +records[r].fields.clone();                                   
      1037 +                        let names_match =                    
      1038 +                            decl.len() == fields.len() && dec
           +l.iter().zip(fields).all(|((d, _), (f, _, _))| d == f);      
      1039 +                        if !names_match {                    
      1040 +                            let want: Vec<&str> = decl.iter()
           +.map(|f| f.0.as_str()).collect();                            
      1041 +                            self.error(*pos, format!("record 
           +'{tname}' needs exactly the fields {{{}}} in that order", wan
           +t.join(", ")));                                              
      1042 +                        } else {                             
      1043 +                            for ((fname, fty), (_, aty)) in d
           +ecl.iter().zip(&tfields) {                                   
      1044 +                                if !self.fits(*aty, *fty) {  
      1045 +                                    let (w, g) = (self.type_n
           +ame(*fty), self.type_name(*aty));                            
      1046 +                                    self.error(*pos, format!(
           +"field '{fname}' of '{tname}' must be {w}, found {g}"));     
      1047 +                                }                            
      1048 +                            }                                
      1049 +                        }                                    
      1050 +                        Self::te(TK::Record(r, tfields.into_i
           +ter().map(|f| f.0).collect()), t, pos.0)                     
      1051 +                    }                                        
      1052 +                    Type::Error => Self::te(TK::Nil, Type::Er
           +ror, pos.0),                                                 
      1053 +                    _ => {                                   
      1054 +                        self.error(*pos, format!("'{tname}' i
           +s not a record type"));                                      
      1055 +                        Self::te(TK::Nil, Type::Error, pos.0)
      1056 +                    }                                        
      1057 +                }                                            
      1058 +            }                                                
      1059 +            Exp::Seq(es) => {                                
      1060 +                let mut items = Vec::new();                  
      1061 +                let mut ty = Type::Unit;                     
      1062 +                for e in es {                                
      1063 +                    let (t, ety) = self.exp(e, line);        
      1064 +                    items.push(t);                           
      1065 +                    ty = ety;                                
      1066 +                }                                            
      1067 +                let l = items.first().map_or(line, |t| t.line
           +);                                                           
      1068 +                Self::te(TK::Seq(items), ty, l)              
      1069 +            }                                                
      1070 +            Exp::Assign(v, rhs, pos) => {                    
      1071 +                if let Var::Simple(name, vpos) = v {         
      1072 +                    if let Some(VEntry::Var { readonly: true,
           + .. }) = self.venv.get(name) {                               
      1073 +                        self.error(*vpos, format!("cannot ass
           +ign to loop variable '{name}'"));                            
      1074 +                    }                                        
      1075 +                }                                            
      1076 +                let (tv, vty) = self.var(v);                 
      1077 +                let (tr, rty) = self.exp(rhs, pos.0);        
      1078 +                if !self.fits(rty, vty) {                    
      1079 +                    let (w, g) = (self.type_name(vty), self.t
           +ype_name(rty));                                              
      1080 +                    self.error(*pos, format!("cannot assign {
           +g} to a variable of {w}"));                                  
      1081 +                }                                            
      1082 +                let rhs = Box::new(tr);                      
      1083 +                let k = match tv.k {                         
      1084 +                    TK::Get(d) => TK::Set(d, rhs),           
      1085 +                    TK::Field(r, i) => TK::SetField(r, i, rhs
           +),                                                           
      1086 +                    TK::Sub(a, i) => TK::SetSub(a, i, rhs),  
      1087 +                    // Assigning to a variable of unit type: 
           +just evaluate.                                               
      1088 +                    _ => TK::Seq(vec![*rhs]),                
      1089 +                };                                           
      1090 +                Self::te(k, Type::Unit, pos.0)               
      1091 +            }                                                
      1092 +            Exp::If(c, t, e, pos) => {                       
      1093 +                let (tc, cty) = self.exp(c, pos.0);          
      1094 +                self.check_int(cty, *pos, "condition of 'if'"
           +);                                                           
      1095 +                let (tt, tty) = self.exp(t, pos.0);          
      1096 +                match e {                                    
      1097 +                    None => {                                
      1098 +                        if self.actual(tty) != Type::Unit && 
           +tty != Type::Error {                                         
      1099 +                            let found = self.type_name(tty); 
      1100 +                            self.error(*pos, format!("'if-the
           +n' without 'else' must not produce a value (found {found})"))
           +;                                                            
      1101 +                        }                                    
      1102 +                        Self::te(TK::If(Box::new(tc), Box::ne
           +w(tt), None), Type::Unit, pos.0)                             
      1103 +                    }                                        
      1104 +                    Some(e) => {                             
      1105 +                        let (te, ety) = self.exp(e, pos.0);  
      1106 +                        let ty = if self.fits(tty, ety) && se
           +lf.actual(tty) != Type::Nil {                                
      1107 +                            tty                              
      1108 +                        } else if self.fits(ety, tty) {      
      1109 +                            ety                              
      1110 +                        } else if self.actual(tty) == Type::N
           +il && self.actual(ety) == Type::Nil {                        
      1111 +                            Type::Nil                        
      1112 +                        } else {                             
      1113 +                            let (a, b) = (self.type_name(tty)
           +, self.type_name(ety));                                      
      1114 +                            self.error(*pos, format!("branche
           +s of 'if' have different types: {a} and {b}"));              
      1115 +                            Type::Error                      
      1116 +                        };                                   
      1117 +                        let (k, ty) = (TK::If(Box::new(tc), B
           +ox::new(tt), Some(Box::new(te))), ty);                       
      1118 +                        // A value-producing branch whose typ
           +e is "error" still                                           
      1119 +                        // pushes; keep `value` consistent wi
           +th the branches.                                             
      1120 +                        let value = self.actual(ty) != Type::
           +Unit;                                                        
      1121 +                        (TE { k, value, line: pos.0 }, ty)   
      1122 +                    }                                        
      1123 +                }                                            
      1124 +            }                                                
      1125 +            Exp::While(c, b, pos) => {                       
      1126 +                let (tc, cty) = self.exp(c, pos.0);          
      1127 +                self.check_int(cty, *pos, "condition of 'whil
           +e'");                                                        
      1128 +                self.loop_depth += 1;                        
      1129 +                let (tb, bty) = self.exp(b, pos.0);          
      1130 +                self.loop_depth -= 1;                        
      1131 +                if self.actual(bty) != Type::Unit && bty != T
           +ype::Error {                                                 
      1132 +                    let found = self.type_name(bty);         
      1133 +                    self.error(*pos, format!("body of 'while'
           + must not produce a value (found {found})"));                
      1134 +                }                                            
      1135 +                Self::te(TK::While(Box::new(tc), Box::new(tb)
           +), Type::Unit, pos.0)                                        
      1136 +            }                                                
      1137 +            Exp::For(v, lo, hi, b, pos) => {                 
      1138 +                let (tlo, loty) = self.exp(lo, pos.0);       
      1139 +                self.check_int(loty, *pos, "lower bound of 'f
           +or'");                                                       
      1140 +                let (thi, hity) = self.exp(hi, pos.0);       
      1141 +                self.check_int(hity, *pos, "upper bound of 'f
           +or'");                                                       
      1142 +                self.venv.push();                            
      1143 +                let d = self.new_decl();                     
      1144 +                self.venv.insert(v, VEntry::Var { decl: d, ty
           +: Type::Int, readonly: true });                              
      1145 +                self.loop_depth += 1;                        
      1146 +                let (tb, bty) = self.exp(b, pos.0);          
      1147 +                self.loop_depth -= 1;                        
      1148 +                self.venv.pop();                             
      1149 +                if self.actual(bty) != Type::Unit && bty != T
           +ype::Error {                                                 
      1150 +                    let found = self.type_name(bty);         
      1151 +                    self.error(*pos, format!("body of 'for' m
           +ust not produce a value (found {found})"));                  
      1152 +                }

           +hi), Box::new(tb)), Type::Unit, pos.0)                       
      1154 +            }                                                
      1155 +            Exp::Break(pos) => {                             
      1156 +                if self.loop_depth == 0 {                    
      1157 +                    self.error(*pos, "'break' outside of a lo
           +op");                                                        
      1158 +                }                                            
      1159 +                Self::te(TK::Break, Type::Unit, pos.0)       
      1160 +            }                                                
      1161 +            Exp::Let(decs, body, pos) => self.let_exp(decs, b
           +ody, *pos, false),

      1163 +                let t = self.lookup_type(tname, *pos);       
      1164 +                let (ts, sty) = self.exp(size, pos.0);       
      1165 +                self.check_int(sty, *pos, "array size");     
      1166 +                let (ti, ity) = self.exp(init, pos.0);       
      1167 +                match self.actual(t) {                       
      1168 +                    Type::Array(a) => {                      
      1169 +                        let elem = self.arrays[a].elem;      
      1170 +                        if !self.fits(ity, elem) {           
      1171 +                            let (w, g) = (self.type_name(elem
           +), self.type_name(ity));                                     
      1172 +                            self.error(*pos, format!("initial
           + value of '{tname}' elements must be {w}, found {g}"));      
      1173 +                        }                                    
      1174 +                        Self::te(TK::Array(Box::new(ts), Box:
           +:new(ti)), t, pos.0)                                         
      1175 +                    }                                        
      1176 +                    Type::Error => Self::te(TK::Nil, Type::Er
           +ror, pos.0),                                                 
      1177 +                    _ => {                                   
      1178 +                        self.error(*pos, format!("'{tname}' i
           +s not an array type"));                                      
      1179 +                        Self::te(TK::Nil, Type::Error, pos.0)
      1180 +                    }                                        
      1181 +                }                                            
      1182 +            }                                                
      1183 +        }                                                    
      1184 +    }

      1186 +    fn var(&mut self, v: &Var) -> (TE, Type) {               
      1187 +        match v {

           +e) {

      1190 +                    self.use_decl(decl);                     
      1191 +                    if self.actual(ty) == Type::Unit {       
      1192 +                        // A variable of type unit holds noth
           +ing worth loading.

           +e: false, line: pos.0 }, ty);                                
      1194 +                    }

      1196 +                }

      1198 +                    self.error(*pos, format!("'{name}' is a f

      1200 +                }

           +ble '{name}'"));

           +s.0)                                                         
      1218 +                        }                                    
      1219 +                    },

           +ror, pos.0),

      1222 +                        let n = self.type_name(other);       
      1223 +                        self.error(*pos, format!("cannot acce
           +ss field '{f}' of {n}: not a record"));                      
      1224 +                        Self::te(TK::Nil, Type::Error, pos.0)
      1225 +                    }                                        
      1226 +                }                                            
      1227 +            }                                                
      1228 +            Var::Subscript(a, i, pos) => {                   
      1229 +                let (ta, aty) = self.var(a);                 
      1230 +                let (ti, ity) = self.exp(i, pos.0);          
      1231 +                self.check_int(ity, *pos, "array index");    
      1232 +                match self.actual(aty) {                     
      1233 +                    Type::Array(ai) => {                     
      1234 +                        let elem = self.arrays[ai].elem;     
      1235 +                        Self::te(TK::Sub(Box::new(ta), Box::n
           +ew(ti)), elem, pos.0)                                        
      1236 +                    }                                        
      1237 +                    Type::Error => Self::te(TK::Nil, Type::Er
           +ror, pos.0),                                                 
      1238 +                    other => {                               
      1239 +                        let n = self.type_name(other);       
      1240 +                        self.error(*pos, format!("cannot subs
           +cript {n}: not an array"));                                  
      1241 +                        Self::te(TK::Nil, Type::Error, pos.0)
      1242 +                    }                                        
      1243 +                }                                            
      1244 +            }                                                
      1245 +        }                                                    
      1246 +    }

      1248 +    fn let_exp(&mut self, decs: &[Dec], body: &[Exp], pos: Po
           +s, outermost: bool) -> (TE, Type) {                          
      1249 +        self.tenv.push();                                    
      1250 +        self.venv.push();                                    
      1251 +        let mut tdecs = Vec::new();                          
      1252 +        for d in decs {                                      
      1253 +            match d {                                        
      1254 +                Dec::Types(group) => self.type_group(group), 
      1255 +                Dec::Var(name, typ, init, vpos) => {         
      1256 +                    let (ti, ity) = self.exp(init, vpos.0);  
      1257 +                    let ty = match typ {                     
      1258 +                        Some((tn, tpos)) => {                
      1259 +                            let t = self.lookup_type(tn, *tpo
           +s);                                                          
      1260 +                            if !self.fits(ity, t) {          
      1261 +                                let (w, g) = (self.type_name(
           +t), self.type_name(ity));                                    
      1262 +                                self.error(*vpos, format!("va
           +riable '{name}' is declared as {w} but initialised with {g}")
           +);                                                           
      1263 +                            }                                
      1264 +                            t                                
      1265 +                        }                                    
      1266 +                        None => {                            
      1267 +                            if self.actual(ity) == Type::Nil 
           +{                                                            
      1268 +                                self.error(*vpos, format!("va
           +riable '{name}' is initialised with nil, so it needs a declar
           +ed record type"));                                           
      1269 +                            }                                
      1270 +                            ity                              
      1271 +                        }                                    
      1272 +                    };                                       
      1273 +                    let decl = self.new_decl();              
      1274 +                    self.venv.insert(name, VEntry::Var { decl
           +, ty, readonly: false });                                    
      1275 +                    tdecs.push(TD::Var(decl, ti));           
      1276 +                }                                            
      1277 +                Dec::Funs(group) => {                        
      1278 +                    let fs = self.fun_group(group, outermost)
           +;                                                            
      1279 +                    if !fs.is_empty() {                      
      1280 +                        tdecs.push(TD::Funs(fs));            
      1281 +                    }                                        
      1282 +                }                                            
      1283 +            }                                                
      1284 +        }                                                    
      1285 +        let (tb, ty) = self.exp(&Exp::Seq(Vec::new()), pos.0)
           +;                                                            
      1286 +        let _ = tb;                                          
      1287 +        let mut items = Vec::new();                          
      1288 +        let mut bty = ty;                                    
      1289 +        for e in body {                                      
      1290 +            let (t, ety) = self.exp(e, pos.0);               
      1291 +            items.push(t);                                   
      1292 +            bty = ety;                                       
      1293 +        }                                                    
      1294 +        self.venv.pop();                                     
      1295 +        self.tenv.pop();                                     
      1296 +        let value = self.actual(bty) != Type::Unit;          
      1297 +        let seq = TE { k: TK::Seq(items), value, line: pos.0 
           +};                                                           
      1298 +        (TE { k: TK::Let(tdecs, Box::new(seq)), value, line: 
           +pos.0 }, bty)                                                
      1299 +    }                                                        
      1300 +                                                             
      1301 +    fn type_group(&mut self, group: &[(String, Ty, Pos)]) {  
      1302 +        let mut seen = HashSet::new();                       
      1303 +        let mut ids = Vec::new();                            
      1304 +        for (name, _, pos) in group {                        
      1305 +            if !seen.insert(name.as_str()) {                 
      1306 +                self.error(*pos, format!("type '{name}' is de
           +clared twice in the same group of mutually recursive types"))
           +;                                                            
      1307 +            }                                                
      1308 +            self.names.push((name.clone(), None));           
      1309 +            let id = self.names.len() - 1;                   
      1310 +            ids.push(id);                                    
      1311 +            self.tenv.insert(name, Type::Name(id));          
      1312 +        }                                                    
      1313 +        for ((name, ty, _), &id) in group.iter().zip(&ids) { 
      1314 +            let t = match ty {                               
      1315 +                Ty::Name(n, p) => self.lookup_type(n, *p),   
      1316 +                Ty::Array(n, p) => {                         
      1317 +                    let elem = self.lookup_type(n, *p);      
      1318 +                    self.arrays.push(ArrayInfo { name: name.c
           +lone(), elem });                                             
      1319 +                    Type::Array(self.arrays.len() - 1)       
      1320 +                }                                            
      1321 +                Ty::Record(fields) => {                      
      1322 +                    let mut fs = Vec::new();                 
      1323 +                    let mut fseen = HashSet::new();          
      1324 +                    for f in fields {                        
      1325 +                        if !fseen.insert(f.name.as_str()) {  
      1326 +                            self.error(f.pos, format!("field 
           +'{}' appears twice in record type '{name}'", f.name));       
      1327 +                        }                                    
      1328 +                        let t = self.lookup_type(&f.typ, f.po
           +s);                                                          
      1329 +                        fs.push((f.name.clone(), t));        
      1330 +                    }                                        
      1331 +                    self.records.push(RecordInfo { name: name
           +.clone(), fields: fs });                                     
      1332 +                    Type::Record(self.records.len() - 1)     
      1333 +                }                                            
      1334 +            };                                               
      1335 +            self.names[id].1 = Some(t);                      
      1336 +        }                                                    
      1337 +        // Every cycle must pass through a record or array ty
           +pe.                                                          
      1338 +        for ((name, _, pos), &id) in group.iter().zip(&ids) {
      1339 +            let mut t = Type::Name(id);                      
      1340 +            let mut steps = 0;                               
      1341 +            while let Type::Name(n) = t {                    
      1342 +                steps += 1;                                  
      1343 +                if steps > ids.len() {                       
      1344 +                    self.error(*pos, format!("type '{name}' i
           +s defined in terms of itself (cycles must pass through a reco
           +rd or array type)"));                                        
      1345 +                    self.names[id].1 = Some(Type::Error);    
      1346 +                    break;                                   
      1347 +                }                                            
      1348 +                t = self.names[n].1.unwrap_or(Type::Error);  
      1349 +            }                                                
      1350 +        }                                                    
      1351 +    }

      1353 +    fn fun_group(&mut self, group: &[FunDec], outermost: bool
           +) -> Vec<TFun> {                                             
      1354 +        let mut seen = HashSet::new();                       
      1355 +        let mut headers = Vec::new();                        
      1356 +        for f in group {                                     
      1357 +            if !seen.insert(f.name.as_str()) {               
      1358 +                self.error(f.pos, format!("function '{}' is d
           +eclared twice in the same group of mutually recursive functio
           +ns", f.name));                                               
      1359 +            }                                                
      1360 +            let formals: Vec<Type> = f.params.iter().map(|p| 
           +self.lookup_type(&p.typ, p.pos)).collect();                  
      1361 +            let result = match &f.result {                   
      1362 +                Some((t, p)) => self.lookup_type(t, *p),     
      1363 +                None => Type::Unit,                          
      1364 +            };                                               
      1365 +            let callee = if f.body.is_some() {               
      1366 +                FunRef::Local(self.new_decl())               
      1367 +            } else {                                         
      1368 +                FunRef::Extern(f.name.clone())               
      1369 +            };                                               
      1370 +            self.venv.insert(&f.name, VEntry::Fun { callee: c
           +allee.clone(), formals: formals.clone(), result });          
      1371 +            headers.push((callee, formals, result));         
      1372 +        }                                                    
      1373 +        let mut out = Vec::new();                            
      1374 +        for (f, (callee, formals, result)) in group.iter().zi
           +p(headers) {                                                 
      1375 +            let (Some(body), FunRef::Local(decl)) = (&f.body,
           + callee) else { continue };                                  
      1376 +            let saved = (self.cur_fn, self.loop_depth);      
      1377 +            self.cur_fn = self.next_fn;                      
      1378 +            self.next_fn += 1;                               
      1379 +            self.loop_depth = 0;                             
      1380 +            self.venv.push();                                
      1381 +            let mut pseen = HashSet::new();                  
      1382 +            let mut params = Vec::new();                     
      1383 +            for (p, &t) in f.params.iter().zip(&formals) {   
      1384 +                if !pseen.insert(p.name.as_str()) {          
      1385 +                    self.error(p.pos, format!("parameter '{}'
           + appears twice", p.name));                                   
      1386 +                }                                            
      1387 +                let d = self.new_decl();                     
      1388 +                self.venv.insert(&p.name, VEntry::Var { decl:
           + d, ty: t, readonly: false });                               
      1389 +                params.push((d, check_kind(self.actual(t))));
      1390 +            }                                                
      1391 +            let (tb, bty) = self.exp(body, f.pos.0);         
      1392 +            self.venv.pop();                                 
      1393 +            (self.cur_fn, self.loop_depth) = saved;          
      1394 +            if self.actual(result) == Type::Unit {           
      1395 +                if self.actual(bty) != Type::Unit && bty != T
           +ype::Error {                                                 
      1396 +                    let found = self.type_name(bty);         
      1397 +                    self.error(f.pos, format!("procedure '{}'
           + must not produce a value (its body has {found}); declare a r
           +esult type", f.name));                                       
      1398 +                }                                            
      1399 +            } else if !self.fits(bty, result) {              
      1400 +                let (w, g) = (self.type_name(result), self.ty
           +pe_name(bty));                                               
      1401 +                self.error(f.pos, format!("function '{}' must
           + return {w}, but its body has {g}", f.name));                
      1402 +            }                                                
      1403 +            out.push(TFun {                                  
      1404 +                decl,                                        
      1405 +                name: f.name.clone(),                        
      1406 +                params,                                      
      1407 +                body: tb,                                    
      1408 +                returns_value: self.actual(result) != Type::U
           +nit,                                                         
      1409 +                exported: outermost,                         
      1410 +            });                                              
      1411 +        }                                                    
      1412 +        out                                                  
      1413 +    }                                                        
      1414 +}

      1416 +// ----------------------------------------------------------
           +-----------------                                            
      1417 +// Code generation                                           
      1418 +                                                             
      1419 +struct Compiler {                                            
      1420 +    g: Gen,                                                  
      1421 +    captured: HashSet<DeclId>,                               
      1422 +    /// Innermost loop last: (exit label, stack height at the
           + loop head).                                                 
      1423 +    loops: Vec<(Label, i64)>,                                
      1424 +    record_descs: HashMap<usize, u32>,                       
      1425 +    records: Vec<RecordInfo>,                                
      1426 +}                                                            
      1427 +                                                             
      1428 +impl Compiler {                                              
      1429 +    fn declare(&mut self, d: DeclId) -> VarLoc {             
      1430 +        let captured = self.captured.contains(&d);           
      1431 +        self.g.declare(d, captured)                          
      1432 +    }                                                        
      1433 +                                                             
      1434 +    fn loc(&mut self, d: DeclId) -> VarLoc {                 
      1435 +        self.g.resolve(d).expect("checked variable is in scop
           +e")                                                          
      1436 +    }                                                        
      1437 +                                                             
      1438 +    /// Compile for a value even if the expression is of type
           + unit.                                                       
      1439 +    fn value(&mut self, e: &TE) {                            
      1440 +        self.exp(e);                                         
      1441 +        if !e.value {                                        
      1442 +            self.g.emit(Op::Nil);                            
      1443 +        }                                                    
      1444 +    }                                                        
      1445 +                                                             
      1446 +    /// Compile a condition, jumping to `on_false` when it is
           + zero.                                                       
      1447 +    fn cond(&mut self, e: &TE, on_false: Label) {            
      1448 +        if let TK::Cmp(op, a, b) = &e.k {                    
      1449 +            self.exp(a);                                     
      1450 +            self.exp(b);                                     
      1451 +            self.g.b().line = e.line;                        
      1452 +            self.g.emit(*op);                                
      1453 +            self.g.b().jump(Op::JumpIfFalse, on_false);      
      1454 +        } else {                                             
      1455 +            self.exp(e);                                     
      1456 +            self.g.b().jump(Op::JumpIfZero, on_false);       
      1457 +        }                                                    
      1458 +    }                                                        
      1459 +                                                             
      1460 +    fn exp(&mut self, e: &TE) {                              
      1461 +        self.g.b().line = e.line;                            
      1462 +        match &e.k {                                         
      1463 +            TK::Int(i) => {                                  
      1464 +                let k = self.g.pool.int(*i);                 
      1465 +                self.g.emit(Op::Const(k));                   
      1466 +            }                                                
      1467 +            TK::Str(s) => {                                  
      1468 +                let k = self.g.pool.str(s);                  
      1469 +                self.g.emit(Op::Const(k));                   
      1470 +            }                                                
      1471 +            TK::Nil => self.g.emit(Op::Nil),                 
      1472 +            TK::Get(d) => {                                  
      1473 +                let loc = self.loc(*d);                      
      1474 +                self.g.emit_get(loc);                        
      1475 +            }                                                
      1476 +            TK::Set(d, v) => {                               
      1477 +                self.value(v);                               
      1478 +                let loc = self.loc(*d);                      
      1479 +                self.g.emit_set(loc);                        
      1480 +            }                                                
      1481 +            TK::Field(r, i) => {                             
      1482 +                self.exp(r);                                 
      1483 +                self.g.b().line = e.line;                    
      1484 +                self.g.emit(Op::GetField(*i));               
      1485 +            }                                                
      1486 +            TK::SetField(r, i, v) => {                       
      1487 +                self.exp(r);                                 
      1488 +                self.value(v);                               
      1489 +                self.g.b().line = e.line;                    
      1490 +                self.g.emit(Op::SetField(*i));               
      1491 +            }                                                
      1492 +            TK::Sub(a, i) => {                               
      1493 +                self.exp(a);                                 
      1494 +                self.exp(i);                                 
      1495 +                self.g.b().line = e.line;                    
      1496 +                self.g.emit(Op::ArrayGet);                   
      1497 +            }                                                
      1498 +            TK::SetSub(a, i, v) => {                         
      1499 +                self.exp(a);                                 
      1500 +                self.exp(i);                                 
      1501 +                self.value(v);                               
      1502 +                self.g.b().line = e.line;                    
      1503 +                self.g.emit(Op::ArraySet);                   
      1504 +            }                                                
      1505 +            TK::Call(callee, args) => {                      
      1506 +                match callee {                               
      1507 +                    Callee::Local(d) => {                    
      1508 +                        let loc = self.loc(*d);              
      1509 +                        self.g.emit_get(loc);                
      1510 +                    }                                        
      1511 +                    Callee::Native(id) => self.g.emit(Op::Nat
           +ive(*id as u32)),                                            
      1512 +                    Callee::Extern(name, _) => {             
      1513 +                        let k = self.g.pool.str(name);       
      1514 +                        self.g.emit(Op::GetGlobal(k));       
      1515 +                    }                                        
      1516 +                }                                            
      1517 +                for a in args {                              
      1518 +                    self.value(a);                           
      1519 +                }                                            
      1520 +                self.g.b().line = e.line;                    
      1521 +                self.g.emit(Op::Call(args.len() as u32));    
      1522 +                if let Callee::Extern(_, kind) = callee {    
      1523 +                    match kind {                             
      1524 +                        CheckKind::Int => self.g.emit(Op::Che
           +ckInt),                                                      
      1525 +                        CheckKind::Str => self.g.emit(Op::Che
           +ckStr),                                                      
      1526 +                        CheckKind::None => {}                
      1527 +                    }                                        
      1528 +                }                                            
      1529 +                if !e.value {                                
      1530 +                    self.g.emit(Op::Pop);                    
      1531 +                }                                            
      1532 +            }                                                
      1533 +            TK::Bin(op, a, b) => {                           
      1534 +                self.exp(a);                                 
      1535 +                self.exp(b);                                 
      1536 +                self.g.b().line = e.line;                    
      1537 +                self.g.emit(*op);                            
      1538 +            }                                                
      1539 +            TK::Cmp(op, a, b) => {                           
      1540 +                self.exp(a);                                 
      1541 +                self.exp(b);                                 
      1542 +                self.g.b().line = e.line;                    
      1543 +                self.g.emit(*op);                            
      1544 +                self.g.emit(Op::BoolToInt);                  
      1545 +            }                                                
      1546 +            TK::And(a, b) => {                               
      1547 +                let (f, end) = (self.g.b().new_label(), self.
           +g.b().new_label());                                          
      1548 +                self.cond(a, f);                             
      1549 +                self.exp(b);                                 
      1550 +                self.g.b().jump(Op::Jump, end);              
      1551 +                self.g.b().place(f);                         
      1552 +                let k = self.g.pool.int(0);                  
      1553 +                self.g.emit(Op::Const(k));                   
      1554 +                self.g.b().place(end);                       
      1555 +            }                                                
      1556 +            TK::Or(a, b) => {                                
      1557 +                let (f, end) = (self.g.b().new_label(), self.
           +g.b().new_label());                                          
      1558 +                self.cond(a, f);
